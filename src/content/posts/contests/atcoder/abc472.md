# B - Break a Stick

> 关键词：签到

## 思路

做一个前缀和，枚举分界线即可，时间复杂度 $O(n)$。

# C - On a Diet

> 关键词：模拟

## 思路

注意到日子是一天一天过的，所以用不用滑动窗口均可。本质上就是在模拟。注意 $cur$ 删除值的时候首先要保证前 $m$ 天存在，且需满足 $st[i - m] = true$。

## Code

```c++

using namespace std;

#define eb emplace_back
// clang-format on

using ll = long long;
using ld = long double;
using ull = unsigned long long;
using pii = pair<int, int>;
using pdd = pair<double, double>;
using pll = pair<long long, long long>;
using i128 = __int128;

const int dx[] = {-1, 0, 1, 0, -1, 1, 1, -1};
const int dy[] = {0, 1, 0, -1, 1, 1, -1, -1};
const int inf = 0x3f3f3f3f;
const int N = 0;
const ll INF = 4e18;

void solve()
{
    ll n, m, k;
    cin >> n >> m >> k;

    vector<ll> a(n + 1);
    for (int i = 1; i <= n; i++)
        cin >> a[i];

    ll cur = 0;
    for (int i = 1; i <= n; i++)
    {
        if (i - m >= 1 && st[i - m] == true)
            cur -= a[i - m];

        if (cur + a[i] <= k)
            cur += a[i], st[i] = true;
    }

    for (int i = 1; i <= n; i++)
        cout << (st[i] ? "Yes" : "No") << endl;
}

int main()
{
    fastio();

    int T = 1;
    // cin >> T;

    while (T--)
        solve();
```

# D - Bomber Mad

> 关键词：bfs

## 思路

我们首先处理出哪些行和列上有炸弹，然后将那些安全的点都放进队列里做长度最大为 $k$ 的 $bfs$。

注意读清题，在 $bfs$ 的过程中，只需要走到的格子上没有炸弹即可。

## Code

```c++
// Problem: AT ABC472 D
// Contest: AtCoder - AtCoder Beginner Contest 472
// URL: https://atcoder.jp/contests/abc472/tasks/abc472_d
// Time: 2026-08-22 21:07:56
#include <bits/stdc++.h>
using namespace std;

// clang-format off
#define endl '\n'
#define all(x) (x).begin(), (x).end()
#define fastio() ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
#define eb emplace_back
// clang-format on

using ll = long long;
using ld = long double;
using ull = unsigned long long;
using pii = pair<int, int>;
using pdd = pair<double, double>;
using pll = pair<long long, long long>;
using i128 = __int128;

const int dx[] = {-1, 0, 1, 0, -1, 1, 1, -1};
const int dy[] = {0, 1, 0, -1, 1, 1, -1, -1};
const int inf = 0x3f3f3f3f;
const int N = 0;
const ll INF = 4e18;

void solve()
{
    int n, m, k;
    cin >> n >> m >> k;

    vector<vector<char>> g(n + 1, vector<char>(m + 1));
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= m; j++)
            cin >> g[i][j];

    vector<bool> row(n + 1, false), col(m + 1, false);
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= m; j++)
            if (g[i][j] == '#')
                row[i] = true, col[j] = true;

    vector<vector<int>> dist(n + 1, vector<int>(m + 1, inf));
    queue<pii> q;

    ll res = 0;
    for (int i = 1; i <= n; i++)
        for (int j = 1; j <= m; j++)
            if (!row[i] && !col[j])
                dist[i][j] = 0, q.push({i, j}), res++;

    while (!q.empty())
    {
        auto [a, b] = q.front();
        q.pop();

        if (dist[a][b] == k)
            continue;

        for (int i = 0; i < 4; i++)
        {
            int x = a + dx[i], y = b + dy[i];
            if (x < 1 || x > n || y < 1 || y > m || g[x][y] == '#')
                continue;

            if (dist[x][y] != inf)
                continue;

            dist[x][y] = dist[a][b] + 1;
            q.push({x, y});
            res++;
        }
    }

    cout << res << endl;
}

int main()
{
    fastio();

    int T = 1;
    // cin >> T;

    while (T--)
        solve();

    return 0;
}

```

# E - Odd Cycle

> 关键词：二分图，染色法

## 思路

看到判断是否存在奇数环，首先考虑二分图判定。我们知道二分图中是一定不含奇数环的，所以使用染色法对整张图染色。

考虑如何找出一个奇环。我们记录每个节点的父亲，当枚举到两个节点 $u,v$ 颜色相同时，说明 $u,v$ 是奇环的两个端点。我们从 $v$ 开始沿着祖先不断向上跳，直到跳到 $u$ 即可。

## Code

```c++
// Problem: AT ABC472 E
// Contest: AtCoder - AtCoder Beginner Contest 472
// URL: https://atcoder.jp/contests/abc472/tasks/abc472_e
// Time: 2026-08-22 21:30:00
#include <bits/stdc++.h>
using namespace std;

// clang-format off
#define endl '\n'
#define all(x) (x).begin(), (x).end()
#define fastio() ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
#define eb emplace_back
// clang-format on

using ll = long long;
using ld = long double;
using ull = unsigned long long;
using pii = pair<int, int>;
using pdd = pair<double, double>;
using pll = pair<long long, long long>;
using i128 = __int128;

const int dx[] = {-1, 0, 1, 0, -1, 1, 1, -1};
const int dy[] = {0, 1, 0, -1, 1, 1, -1, -1};
const int inf = 0x3f3f3f3f;
const int N = 0;
const ll INF = 4e18;

void solve()
{
    int n, m;
    cin >> n >> m;

    vector<vector<int>> g(n + 1);
    for (int i = 1, u, v; i <= m; i++)
        cin >> u >> v, g[u].eb(v), g[v].eb(u);

    vector<int> color(n + 1, -1), fa(n + 1, 0), res;
    color[1] = 0;

    auto dfs = [&](auto &&self, int u, int p) -> void
    {
        fa[u] = p;
        for (auto v: g[u])
        {
            if (v == p)
                continue;

            if (color[v] == -1)
            {
                color[v] = color[u] ^ 1;
                self(self, v, u);
            }
            else if (color[v] == color[u])
            {
                if (res.empty())
                {
                    int x = u;
                    res.eb(x);
                    while (x != v)
                        x = fa[x], res.eb(x);
                }
            }
        }
    };

    dfs(dfs, 1, 0);
    if (res.empty())
        cout << -1 << endl;
    else
    {
        cout << res.size() << endl;
        for (auto x: res)
            cout << x << ' ';
        cout << endl;
    }
}

int main()
{
    fastio();

    int T = 1;
    cin >> T;

    while (T--)
        solve();

    return 0;
}

```

# F - Centroid of a Slice

> 关键词：计算几何，多边形重心

## 思路

重心公式：${ G_x= \frac{ \sum_i(x_i+x_{i+1})c_i }{ 3S } }$，${ G_y= \frac{ \sum_i(y_i+y_{i+1})c_i }{ 3S } }$。

我们看作每一条边贡献了三个东西：${ \begin{aligned} A_i &= P_i\times P_{i+1}\\ X_i &= (x_i+x_{i+1})A_i\\ Y_i &= (y_i+y_{i+1})A_i \end{aligned} }$，那么 $G$ 就是 $G= \left( \frac{\sum X_i}{3\sum A_i}, \frac{\sum Y_i}{3\sum A_i} \right)$。

然后我们发现，**它是可以前缀和的**。对于询问 $u<v$，子多边形是 $P_u,P_{u+1},\dots,P_v$。它的边分别是 $P_u\to P_{u+1}$，$P_{u+1}\to P_{u+2}$，$\cdots$，$P_{v-1}\to P_v$，还有最后额外的一条 ${P_v\to P_u}$。前面所有边都是原多边形的边，直接用区间求和算它们的贡献即可。然后补上最后一条边的叉积。

对于 $u>v$ 的询问，我们直接复制一遍点，如果 $v<u$，就把 $v+=n$。这里无需考虑前缀和的影响。

> [!NOTE]
>
> 对重心公式认识不够深刻，实际上是应该可以观察到前缀和的。

## Code

```c++
// Problem: AT ABC472 F
// Contest: AtCoder - AtCoder Beginner Contest 472
// URL: https://atcoder.jp/contests/abc472/tasks/abc472_f
// Time: 2026-08-22 21:21:19
#include <bits/stdc++.h>
using namespace std;

// clang-format off
#define endl '\n'
#define all(x) (x).begin(), (x).end()
#define fastio() ios::sync_with_stdio(0); cin.tie(0); cout.tie(0);
#define eb emplace_back
// clang-format on

using ll = long long;
using ld = long double;
using ull = unsigned long long;
using pii = pair<int, int>;
using pdd = pair<double, double>;
using pll = pair<long long, long long>;
using i128 = __int128;

const int dx[] = {-1, 0, 1, 0, -1, 1, 1, -1};
const int dy[] = {0, 1, 0, -1, 1, 1, -1, -1};
const int inf = 0x3f3f3f3f;
const int N = 0;
const ll INF = 4e18;

template<class T>
struct Point
{
    T x, y;

    Point(const T &x = 0, const T &y = 0) : x(x), y(y) {}

    template<class U>
    operator Point<U>() const
    {
        return Point<U>(U(x), U(y));
    }

    Point &operator+=(const Point &other)
    {
        x += other.x;
        y += other.y;
        return *this;
    }

    Point &operator-=(const Point &other)
    {
        x -= other.x;
        y -= other.y;
        return *this;
    }

    Point &operator*=(const T &k)
    {
        x *= k;
        y *= k;
        return *this;
    }

    Point &operator/=(const T &k)
    {
        x /= k;
        y /= k;
        return *this;
    }

    Point operator-() const
    {
        return {-x, -y};
    }

    friend Point operator+(Point a, const Point &b)
    {
        return a += b;
    }

    friend Point operator-(Point a, const Point &b)
    {
        return a -= b;
    }

    friend Point operator*(Point a, const T &k)
    {
        return a *= k;
    }

    friend Point operator*(const T &k, Point a)
    {
        return a *= k;
    }

    friend Point operator/(Point a, const T &k)
    {
        return a /= k;
    }

    friend bool operator==(const Point &a, const Point &b)
    {
        return a.x == b.x && a.y == b.y;
    }

    friend bool operator!=(const Point &a, const Point &b)
    {
        return !(a == b);
    }

    friend istream &operator>>(istream &is, Point &p)
    {
        return is >> p.x >> p.y;
    }

    friend ostream &operator<<(ostream &os, const Point &p)
    {
        return os << "(" << p.x << ", " << p.y << ")";
    }
};

template<class T>
T cross(const Point<T> &a, const Point<T> &b)
{
    return a.x * b.y - a.y * b.x;
}

void solve()
{
    int n, q;
    cin >> n >> q;
    vector<Point<ll>> p(2 * n + 2);

    for (int i = 1; i <= n; i++)
        cin >> p[i], p[i + n] = p[i];

    vector<ll> prea(2 * n + 2), prex(2 * n + 2), prey(2 * n + 2);
    for (int i = 1; i <= 2 * n; i++)
    {
        ll cr = cross(p[i], p[i + 1]);
        prea[i] = prea[i - 1] + cr;
        prex[i] = prex[i - 1] + (p[i].x + p[i + 1].x) * cr;
        prey[i] = prey[i - 1] + (p[i].y + p[i + 1].y) * cr;
    }

    cout << fixed << setprecision(15);
    while (q--)
    {
        int u, v;
        cin >> u >> v;

        if (u > v)
            v += n;

        ll s = prea[v - 1] - prea[u - 1];
        ll x = prex[v - 1] - prex[u - 1];
        ll y = prey[v - 1] - prey[u - 1];

        ll cr = cross(p[v], p[u]);
        s += cr, x += (p[v].x + p[u].x) * cr, y += (p[v].y + p[u].y) * cr;

        ld posx = (ld) x / (3.00 * (ld) s), posy = (ld) y / (3.00 * (ld) s);
        cout << posx << ' ' << posy << endl;
    }
}

int main()
{
    fastio();

    int T = 1;
    // cin >> T;

    while (T--)
        solve();

    return 0;
}
```
